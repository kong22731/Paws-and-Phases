# แก้การลบพื้นหลังด้วยมือ สำหรับ tools/make-sprites.ps1
# ใช้เมื่อมีช่องสีขาวที่ถูกล้อมไว้ (เช่น ระหว่างขา) ที่สคริปต์ตัดสินผิด
#
# รูปแบบ: <key> <remove|keep> <x> <y>
#   key    = ชื่อร่าง เช่น dog_playful_2, cat_sleepy_4
#   remove = บังคับลบช่องสีขาวที่จุดนี้อยู่ / keep = บังคับเก็บไว้ (เช่น ฟองหรือเมฆสีขาว)
#   x y    = ตำแหน่งในรูปต้นฉบับ เป็นสัดส่วน 0-1 (0 0 = มุมซ้ายบน, 1 1 = มุมขวาล่าง)
# หรือ:   <key> pockets <n>
#   ลบทุกช่องสีขาวที่ถูกล้อมไว้ในรูปนี้ ที่ขอบมีเงาสีอ่อนไม่เกิน n% (เหมาะกับรูปที่มีเส้นพลิ้ว ๆ เยอะ เช่น สาย Sleepy)
#
# ดูผลได้ด้วย: powershell -ExecutionPolicy Bypass -File tools\make-sprites.ps1 -Review
#   จะแสดงรายการช่องสีขาวขนาดใหญ่พร้อมตำแหน่ง x y ให้คัดลอกมาใส่ที่นี่
#   แล้วเปิดรูปใน assets/sprites/review (ฟ้า = ช่องที่ลบ, เขียว = สีขาวที่เก็บไว้)

# หรือ:   <key> strict
#   ลบเฉพาะพื้นขาวล้วน (ใช้กับรูปที่มีฟอง/เมฆสีอ่อนติดขอบภาพ ซึ่งปกติจะถูกลบไปด้วย)

# หรือ:   <key> keepshadow
#   ไม่ต้องลบเงาพื้นใต้เท้า (ปกติสคริปต์จะลบเงาสีชมพู/เบจ/เทาที่ส่วนล่างของภาพออก)
